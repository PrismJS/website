Full example:


; calculate all 16 bit Fibonacci numbers

BUFFER = $1000

fibonacci:
   rep   #$30
   clc
   cld
   lda   #1
   sta   BUFFER+2
   dec   a
   stz   BUFFER
@loop:
   lda   BUFFER,x
   inx
   inx
   adc   BUFFER,x
   bcs   @done
   inx
   inx
   sta   BUFFER,x
   dex
   dex
   bne   @loop
@done:
   rts
